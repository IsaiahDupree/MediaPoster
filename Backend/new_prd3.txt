2) Generate scene clips with Sora (programmatically)

OpenAI’s Video API is async: you POST /v1/videos to start a job, poll GET /v1/videos/{id} (or use webhooks), then download with GET /v1/videos/{id}/content. OpenAI Platform

Node/TS-ish example (server-side)

import OpenAI from "openai";
import fs from "node:fs";

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

async function generateSceneClip({
  prompt,
  seconds,
  size = "1280x720",
  referenceImageFileIds = [], // optional
}: {
  prompt: string;
  seconds: number;
  size?: string;
  referenceImageFileIds?: string[];
}) {
  // 1) Start job (POST /v1/videos)
  const job = await openai.videos.create({
    model: "sora-2",          // or "sora-2-pro" for higher fidelity :contentReference[oaicite:1]{index=1}
    prompt,
    seconds,
    size,
    // If you have reference inputs, include them (the guide calls out optional reference inputs/remix IDs) :contentReference[oaicite:2]{index=2}
    // input_image: referenceImageFileIds ...
  });

  // 2) Poll until completed
  let status = job.status;
  let videoId = job.id;

  while (status !== "completed" && status !== "failed") {
    await new Promise((r) => setTimeout(r, 2000));
    const updated = await openai.videos.retrieve(videoId);
    status = updated.status;
  }

  if (status === "failed") throw new Error("Sora job failed");

  // 3) Download MP4 (GET /v1/videos/{id}/content)
  const content = await openai.videos.content(videoId); // returns a stream/bytes depending on SDK version
  return content;
}
Sora 2 / Sora 2 Pro take text or image inputs and output video with synced audio. OpenAI Platform+1

Approach A (simplest): Characters are baked into each scene

You keep character consistency by using a character bible:

consistent style tokens (“flat 2D explainer, minimal shading, high contrast”)

Approach B (more “asset-like”): Generate character-only clips → key them → overlay in Remotion

Workflow:

Generate a character clip on solid green (or solid magenta) background (walk cycle, pointing, idle loop).

In Remotion, overlay that transparent character video on top of any background scene.

That gives you real “assets” you can reuse across tons of videos.

// makeStoryIR.ts
import { ContentBriefV1, StoryIRV1, TrendItemV1 } from "./types";

export function makeStoryIR(trend: TrendItemV1, brief: ContentBriefV1): StoryIRV1 {
  const maxSeconds = brief.constraints?.maxSeconds ?? 58;
  const angle = trend.angleCandidates[0] ?? "most practical angle";

  // Minimal deterministic beat plan (you can later swap with an LLM planner)
  const beats: StoryIRV1["beats"] = [];

  beats.push({
    id: "beat_hook",
    type: "HOOK",
    duration_s: 2.5,
    narration: If you're trying to ${brief.promise.toLowerCase()}, do this first.,
    on_screen: { headline: ${brief.promise}, sub: trend.topic },
    broll: [{ intent: "abstract", query: ${trend.topic} simple icon animation }],
    audio: { music_energy: "high", sfx: ["whoosh"] },
  });

  beats.push({
    id: "beat_promise",
    type: "PROMISE",
    duration_s: 3.0,
    narration: In the next 45 seconds, you'll learn the exact steps that actually work.,
    on_screen: { headline: The 3-step plan },
    broll: [{ intent: "diagram", query: 3 step process diagram minimalist }],
    audio: { music_energy: "mid" },
  });

  const steps = brief.keyPoints.slice(0, 8);
  steps.forEach((kp, i) => {
    beats.push({
      id: beat_step_${i + 1},
      type: "STEP",
      duration_s: 5.0,
      narration: Step ${i + 1}: ${kp}.,
      on_screen: { label: Step ${i + 1}, bullet: kp },
      broll: [{ intent: "ui-demo", query: ${trend.topic} ${kp} UI example }],
      audio: { music_energy: "mid", sfx: ["click"] },
    });
  });

  if (brief.CTA?.text) {
    beats.push({
      id: "beat_cta",
      type: "CTA",
      duration_s: 2.5,
      narration: brief.CTA.text,
      on_screen: { headline: brief.CTA.text },
      audio: { music_energy: "high", sfx: ["pop"] },
    });
  }

  // Clamp total duration
  let total = beats.reduce((s, b) => s + b.duration_s, 0);
  while (total > maxSeconds && beats.length > 3) {
    const removed = beats.splice(beats.findIndex(b => b.type === "STEP"), 1)[0];
    if (!removed) break;
    total = beats.reduce((s, b) => s + b.duration_s, 0);
  }

  return {
    meta: { fps: 30, aspect: "9:16", language: "en", tone: "witty-direct", maxSeconds },
    variables: { topic: trend.topic, angle, audience: brief.audience, promise: brief.promise },
    beats,
  };
}

// makeShotPlan.ts
import crypto from "node:crypto";
import { FormatPackV1, ShotPlanV1, StoryIRV1 } from "./types";

function sha256(s: string) {
  return crypto.createHash("sha256").update(s).digest("hex");
}

export function makeShotPlan(
  ir: StoryIRV1,
  format: FormatPackV1,
  opts?: { model?: "sora-2" | "sora-2-pro"; size?: "720x1280" | "1280x720"; referenceFileIds?: string[] }
): ShotPlanV1 {
  const model = opts?.model ?? "sora-2";
  const size = opts?.size ?? (ir.meta.aspect === "9:16" ? "720x1280" : "1280x720");

  const globalTokens = [
    "clean flat 2D explainer",
    "high contrast",
    "big readable typography space",
    "simple stick figure character",
  ];

  const negativeTokens = ["tiny text", "photorealistic", "busy background"];

  const shots = ir.beats
    .filter(b => format.renderStrategy.soraBeatTypes.includes(b.type))
    .map(b => {
      const beatPrompt = [
        Beat type: ${b.type}.,
        Narration context: ${b.narration},
        b.on_screen?.headline ? On-screen headline: ${b.on_screen.headline} : "",
        b.on_screen?.bullet ? On-screen bullet: ${b.on_screen.bullet} : "",
        "Composition: keep safe margins for captions.",
      ].filter(Boolean).join(" ");

      const prompt = ${globalTokens.join(", ")}. ${beatPrompt}. Avoid: ${negativeTokens.join(", ")}.;

      const cacheKey = sha256(
        JSON.stringify({
          model,
          size,
          prompt,
          refs: opts?.referenceFileIds ?? [],
        })
      );

      return {
        id: shot_${b.id},
        fromBeatId: b.id,

  const res = await fetch(url, {

  // Docs: POST /videos returns {id,status,...} :contentReference[oaicite:6]{index=6}

  });

}

async function pollVideoJob(apiKey: string, id: string) {
  while (true) {
    const res = await openaiFetch(${OPENAI_BASE}/videos/${id}, apiKey, { method: "GET" });
    const job = (await res.json()) as any;
    if (job.status === "completed") return job;
    if (job.status === "failed") throw new Error(Video job failed: ${id});
    await sleep(1500);
  }
}

  // API ref: content download is provided for completed videos :contentReference[oaicite:8]{index=8}
  const res = await openaiFetch(${OPENAI_BASE}/videos/${id}/content, apiKey, { method: "GET" });
  const buf = Buffer.from(await res.arrayBuffer());
  return buf;
}

export async function runSoraShotPlan(opts: {
  apiKey: string;
  shotPlan: ShotPlanV1;
  outDir: string;        // local cache dir (later upload to R2/S3)
  publicBaseUrl?: string; // if you upload elsewhere, swap src to that URL
  concurrency?: number;
}): Promise<AssetManifestV1> {
  const { apiKey, shotPlan, outDir } = opts;
  const concurrency = opts.concurrency ?? 3;
  const limit = pLimit(concurrency);

  await fs.mkdir(outDir, { recursive: true });

  const clips = await Promise.all(
    shotPlan.shots.map(shot =>
      limit(async () => {
        const filename = ${shot.cacheKey}.mp4;
        const filePath = path.join(outDir, filename);

        // Cache hit?

          await fs.stat(filePath);
          return {
            shotId: shot.id,
            beatId: shot.fromBeatId,
            src: opts.publicBaseUrl ? ${opts.publicBaseUrl}/${filename} : filePath,
            seconds: shot.seconds,
            hasAudio: true, // Sora 2 outputs synced audio :contentReference[oaicite:9]{index=9}
          };
        } catch {}

        const job = await createVideoJob(apiKey, shot, shotPlan.references?.file_ids);
        const done = await pollVideoJob(apiKey, job.id);
        const mp4 = await downloadVideoContent(apiKey, job.id);

        await fs.writeFile(filePath, mp4);

        return {
          shotId: shot.id,
          beatId: shot.fromBeatId,
          src: opts.publicBaseUrl ? ${opts.publicBaseUrl}/${filename} : filePath,
          seconds: shot.seconds,
          hasAudio: true,
        };
      })
    )
  );

  return { clips };

// makeRemotionRenderPlan.ts
import { AssetManifestV1, FormatPackV1, RenderPlanRemotionV1, StoryIRV1 } from "./types";

export function makeRemotionRenderPlan(
  ir: StoryIRV1,
  format: FormatPackV1,
  assets: AssetManifestV1,
  outSize?: { w: number; h: number }
): RenderPlanRemotionV1 {
  const fps = ir.meta.fps;
  const size = outSize ?? (ir.meta.aspect === "9:16" ? { w: 720, h: 1280 } : { w: 1280, h: 720 });

  const clipByBeat = new Map(assets.clips.map(c => [c.beatId, c]));

  let cursor = 0;
  const timeline: RenderPlanRemotionV1["timeline"] = [];

  for (const beat of ir.beats) {
    const durationInFrames = Math.max(1, Math.round(beat.duration_s * fps));

    if (format.renderStrategy.soraBeatTypes.includes(beat.type)) {
      const clip = clipByBeat.get(beat.id);
      if (!clip) throw new Error(Missing Sora clip for beat ${beat.id});

      timeline.push({
        id: tl_${beat.id},
        from: cursor,
        durationInFrames,
        kind: "video",
        src: clip.src,
      });
    } else {
      // Native scene rendered by Remotion component
      const componentName = format.componentMap[beat.type] ?? "SceneFallback";
      timeline.push({
        id: tl_${beat.id},
        from: cursor,
        durationInFrames,
        kind: "native",
        componentName,
        props: {
          beat,
          variables: ir.variables,
        },
      });
    }

    cursor += durationInFrames;
  }

  return { meta: { fps, size }, timeline };
}

Segment 2/6 — formatSelector.ts (auto-pick best format)

This is a deterministic scorer. Later you can add an LLM “judge,” but this gets you 80% fast.

// formatSelector.ts
import { ContentBriefV1, TrendItemV1 } from "./types";
import { FORMAT_CATALOG, FormatDescriptor, FormatId } from "./formatCatalog";

type SelectorContext = {
  preferSora?: boolean;          // e.g. you want fully generated scenes
  haveScreenRecord?: boolean;    // devlog format becomes viable if true
  platform?: TrendItemV1["platform"];
};

function scoreFormat(desc: FormatDescriptor, trend: TrendItemV1, brief: ContentBriefV1, ctx: SelectorContext) {
  let score = 0;

  // Platform fit
  const platform = ctx.platform ?? trend.platform;
  if (desc.traits.bestForPlatforms.includes(platform)) score += 3;

  // Goal fit
  if (desc.traits.bestForGoals.includes(brief.goal)) score += 3;

  // Sora preference
  if (ctx.preferSora) {
    score += desc.traits.soraReliance === "high" ? 3 : desc.traits.soraReliance === "mid" ? 1 : -2;
  }

  // Screen-record availability for devlog
  if (!ctx.haveScreenRecord && desc.id === "devlog_screen_native_v1") score -= 4;

  // Content density: more key points -> listicle/devlog (faster pace)
  const kp = brief.keyPoints.length;
  if (kp >= 6) score += desc.traits.pace === "fast" ? 2 : desc.traits.pace === "mid" ? 1 : -1;
  if (kp <= 3) score += desc.traits.pace === "slow" ? 2 : 0;

  // Evidence present -> doc/devlog benefit
  const evidenceCount = trend.evidence?.length ?? 0;
  if (evidenceCount >= 3) score += desc.id === "doc_broll_sora_v1" ? 2 : desc.id === "devlog_screen_native_v1" ? 1 : 0;

  // Hard constraint: max steps
  const maxSteps = desc.pack.rules.constraints.max_steps;
  if (kp > maxSteps) score -= 2;

  return score;
}

export function selectFormat(trend: TrendItemV1, brief: ContentBriefV1, ctx: SelectorContext = {}) {
  const ranked = FORMAT_CATALOG
    .map((d) => ({ d, score: scoreFormat(d, trend, brief, ctx) }))
    .sort((a, b) => b.score - a.score);

  const winner = ranked[0];
  return {
    selectedFormatId: winner.d.id as FormatId,
    format: winner.d,
    ranked: ranked.map((r) => ({ id: r.d.id, label: r.d.label, score: r.score })),
  };
}

// orchestrate.ts
import fs from "node:fs/promises";
import path from "node:path";
import { makeStoryIR } from "./makeStoryIR";
import { makeShotPlan } from "./makeShotPlan";
import { runSoraShotPlan } from "./runSora";
import { makeRemotionRenderPlan } from "./makeRemotionRenderPlan";
import { selectFormat } from "./formatSelector";
import { ContentBriefV1, TrendItemV1 } from "./types";

export async function orchestrateToRenderPlan(opts: {
  trend: TrendItemV1;
  brief: ContentBriefV1;
  apiKey: string;
  outDir: string;
  soraOutDir: string;
  preferSora?: boolean;
  haveScreenRecord?: boolean;
  referenceFileIds?: string[];
}) {
  const { trend, brief } = opts;

  // 1) select format
  const sel = selectFormat(trend, brief, {
    preferSora: opts.preferSora ?? true,
    haveScreenRecord: opts.haveScreenRecord ?? false,
    platform: trend.platform,
  });

  // 2) story IR
  const ir = makeStoryIR(trend, brief);

  // 3) shot plan (only for beats that this format routes to Sora)
  const shotPlan = makeShotPlan(ir, sel.format.pack, {
    model: "sora-2",
    size: ir.meta.aspect === "9:16" ? "720x1280" : "1280x720",
    referenceFileIds: opts.referenceFileIds,
  });

  // 4) run Sora (cache-aware)
  const assets = await runSoraShotPlan({
    apiKey: opts.apiKey,
    shotPlan,
    outDir: opts.soraOutDir,
    concurrency: 3,
  });

  // 5) build Remotion plan (mix sora video + native scenes)
  const plan = makeRemotionRenderPlan(ir, sel.format.pack, assets);

  // 6) write artifacts
  await fs.mkdir(opts.outDir, { recursive: true });
  await fs.writeFile(path.join(opts.outDir, "selected_format.json"), JSON.stringify(sel, null, 2));
  await fs.writeFile(path.join(opts.outDir, "story_ir.json"), JSON.stringify(ir, null, 2));
  await fs.writeFile(path.join(opts.outDir, "shot_plan.json"), JSON.stringify(shotPlan, null, 2));

  await fs.writeFile(path.join(opts.outDir, "render_plan.json"), JSON.stringify(plan, null, 2));

  return { sel, ir, shotPlan, assets, plan };
}

export function buildSoraPrompt(params: {
  shotType: ShotType;
  beatNarration: string;
  onScreen?: { headline?: string; bullet?: string; label?: string };
  globalTokens: string[];
  negativeTokens: string[];
}) {
  const base = ${params.globalTokens.join(", ")}. Context: ${params.beatNarration}.;
  const textHints = [
    params.onScreen?.headline ? Headline: ${params.onScreen.headline} : "",
    params.onScreen?.bullet ? Bullet: ${params.onScreen.bullet} : "",
  ].filter(Boolean).join(" ");

  if (params.shotType === "FULL_SCENE") {
    return ${base} Compose the full scene with character + minimal readable on-screen text. ${textHints} Avoid: ${params.negativeTokens.join(", ")}.;
  }

  if (params.shotType === "BG_ONLY") {
    return ${base} Generate ONLY a clean background plate with no text. Leave empty space for captions and overlays. Avoid: ${params.negativeTokens.join(", ")}.;
  }

  // CHAR_ALPHA (chroma key workflow)
  return ${base} Generate ONLY the character animation on a SOLID GREEN background, no shadows, no gradients, no text. Keep edges clean for chroma key. Avoid: ${params.negativeTokens.join(", ")}.;

Segment 5/6 — Runner upgrade: chroma key postprocess for CHAR_ALPHA

// postprocessAlpha.ts
import { execFile } from "node:child_process";
import { promisify } from "node:util";
const execFileAsync = promisify(execFile);

/**
 * Produces a transparent video (e.g., WebM VP9 with alpha) from a green screen MP4.
 * Requires ffmpeg installed.
 */
export async function chromaKeyToAlpha(opts: {
  inPath: string;
  outPath: string; // recommend .webm for VP9 alpha
  color?: "green" | "magenta";
  similarity?: number; // 0.0-1.0-ish
  blend?: number;      // 0.0-1.0-ish
}) {
  const color = opts.color ?? "green";
  const similarity = opts.similarity ?? 0.18;
  const blend = opts.blend ?? 0.02;

  // ffmpeg filter: colorkey removes key color; format rgba preserves alpha
  // Output: VP9 webm with alpha
  const keyColor = color === "green" ? "0x00FF00" : "0xFF00FF";

  const args = [
    "-y",
    "-i",
    opts.inPath,
    "-vf",
    colorkey=${keyColor}:${similarity}:${blend},format=rgba,
    "-c:v",
    "libvpx-vp9",
    "-pix_fmt",
    "yuva420p",
    "-b:v",
    "2M",
    "-an",
    opts.outPath,
  ];

  await execFileAsync("ffmpeg", args);
}

// validatePlan.ts
import { AssetManifestV2, ShotPlanV2, FormatPackV2 } from "./types.v2";
import { RenderPlanRemotionV1, StoryIRV1 } from "./types";

export function validatePipeline(opts: {
  ir: StoryIRV1;
  format: FormatPackV2;
  shotPlan: ShotPlanV2;
  assets: AssetManifestV2;
  renderPlan: RenderPlanRemotionV1;
}) {
  const { ir, format, shotPlan, assets, renderPlan } = opts;

  // 1) Basic meta alignment
  if (ir.meta.fps !== shotPlan.meta.fps) throw new Error("FPS mismatch: story_ir vs shot_plan");
  if (ir.meta.fps !== renderPlan.meta.fps) throw new Error("FPS mismatch: story_ir vs render_plan");

  // 2) Beat coverage
  const beatIds = new Set(ir.beats.map(b => b.id));
  for (const s of shotPlan.shots) {
    if (!beatIds.has(s.fromBeatId)) throw new Error(Shot references missing beat: ${s.fromBeatId});
  }

  // 3) Sora beats must have shots
  const shotsByBeat = new Map(shotPlan.shots.map(s => [s.fromBeatId, s]));
  for (const b of ir.beats) {
    if (format.renderStrategy.soraBeatTypes.includes(b.type as any)) {
      if (!shotsByBeat.has(b.id)) throw new Error(Missing shot for Sora-routed beat: ${b.id} (${b.type}));
    }
  }

  // 4) Assets must exist for each shot
  const assetsByShot = new Map(assets.clips.map(c => [c.shotId, c]));
  for (const s of shotPlan.shots) {
    const a = assetsByShot.get(s.id);
    if (!a) throw new Error(Missing asset for shot: ${s.id});

    if (s.shotType === "CHAR_ALPHA") {
      if (!a.alphaSrc) throw new Error(CHAR_ALPHA shot missing alphaSrc: ${s.id});
    } else {
      if (!a.src) throw new Error(Non-alpha shot missing src: ${s.id});
    }
  }

  // 5) Render plan sanity: sorted + non-negative durations
  let lastFrom = -1;
  for (const t of renderPlan.timeline) {
    if (t.from < 0) throw new Error(Timeline item has negative 'from': ${t.id});
    if (t.durationInFrames <= 0) throw new Error(Timeline item has non-positive duration: ${t.id});
    if (t.from < lastFrom) throw new Error(Timeline not sorted by 'from' at: ${t.id});
    lastFrom = t.from;
  }

  return { ok: true };
}

Segment 4/6 — Build a layered render plan (BG + CHAR_ALPHA overlay + native overlays)

// makeRemotionRenderPlan.v2.ts
import { RenderPlanRemotionV2 } from "./renderPlan.v2";
import { StoryIRV1 } from "./types";
import { AssetManifestV3 } from "./assetManifest.v3";
import { FormatPackV2 } from "./types.v2";
import { OVERLAY_RULES_V1 } from "./formatOverlayRules";

export function makeRemotionRenderPlanV2(opts: {
  ir: StoryIRV1;
  format: FormatPackV2 & { overlayRules?: typeof OVERLAY_RULES_V1 };
  assets: AssetManifestV3;
  outSize?: { w: number; h: number };
}): RenderPlanRemotionV2 {
  const { ir, format, assets } = opts;
  const fps = ir.meta.fps;
  const size = opts.outSize ?? (ir.meta.aspect === "9:16" ? { w: 720, h: 1280 } : { w: 1280, h: 720 });

  const clipByBeat = new Map(assets.clips.map(c => [c.beatId, c]));

  let cursor = 0;
  const layers: RenderPlanRemotionV2["layers"] = [];
  const overlayRules = format.overlayRules ?? OVERLAY_RULES_V1;

  for (const beat of ir.beats) {
    const durationInFrames = Math.max(1, Math.round(beat.duration_s * fps));
    const from = cursor;

    const isSoraBeat = format.renderStrategy.soraBeatTypes.includes(beat.type as any);
    const clip = clipByBeat.get(beat.id);

    // 1) Background layer
    if (isSoraBeat && clip?.src) {
      layers.push({
        id: bg_${beat.id},
        kind: "VIDEO",
        zIndex: 0,
        from,
        durationInFrames,
        src: clip.src,
        muted: false,
        volume: 1,
      });
    } else {
      // native "background" scene if not Sora-routed
      const componentName = format.componentMap[beat.type] ?? "SceneFallback";
      layers.push({
        id: native_${beat.id},
        kind: "NATIVE",
        zIndex: 0,
        from,
        durationInFrames,
        componentName,
        props: { beat, variables: ir.variables },
      });
    }

    // 2) Character alpha overlay (if present)
    if (clip?.shotType === "CHAR_ALPHA" && clip.alphaSrc) {
      const preset = overlayRules.defaultPreset;
      const placement = overlayRules.presets[preset];

      layers.push({
        id: char_${beat.id},
        kind: "ALPHA_VIDEO",
        zIndex: 10,
        from,
        durationInFrames,
        alphaSrc: clip.alphaSrc,
        transparent: true,
        anchor: placement.anchor,
        transform: placement.transform,
        muted: true, // usually mute alpha overlays
        volume: 0,
      });
    }

    // 3) Native overlay text/captions for Sora beats (optional but common)
    if (isSoraBeat) {
      layers.push({
        id: overlay_text_${beat.id},
        kind: "NATIVE",
        zIndex: 20,
        from,
        durationInFrames,
        componentName: "SceneOverlayText",
        props: { beat, variables: ir.variables },
      });
    }

    cursor += durationInFrames;
  }

  // Sort by time then zIndex (stable render)
  layers.sort((a, b) => (a.from - b.from) || (a.zIndex - b.zIndex));

  return { meta: { fps, size }, layers };
}

// RenderFromPlanV2.tsx
import React from "react";
import { AbsoluteFill, Sequence, Video, staticFile } from "remotion";

// anchor → CSS positioning
function anchorStyle(anchor: Anchor | undefined): React.CSSProperties {
  switch (anchor) {
    case "top-left": return { top: 0, left: 0 };
    case "top": return { top: 0, left: "50%", transform: "translateX(-50%)" };
    case "top-right": return { top: 0, right: 0 };
    case "left": return { top: "50%", left: 0, transform: "translateY(-50%)" };
    case "center": return { top: "50%", left: "50%", transform: "translate(-50%, -50%)" };
    case "right": return { top: "50%", right: 0, transform: "translateY(-50%)" };
    case "bottom-left": return { bottom: 0, left: 0 };
    case "bottom": return { bottom: 0, left: "50%", transform: "translateX(-50%)" };
    case "bottom-right": return { bottom: 0, right: 0 };
    default: return { top: 0, left: 0 };
  }
}

function applyTransform(base: React.CSSProperties, layer: LayerV2): React.CSSProperties {
  const t = layer.transform ?? {};
  const x = t.x ?? 0;
  const y = t.y ?? 0;
  const s = t.scale ?? 1;
  const r = t.rotate ?? 0;
  const o = t.opacity ?? 1;

  // combine with existing transform from anchor
  const existing = base.transform ? ${base.transform}  : "";
  return {
    ...base,
    opacity: o,
    transform: ${existing} translate(${x}px, ${y}px) scale(${s}) rotate(${r}deg),
  };
}

export const RenderFromPlanV2: React.FC<{ plan: RenderPlanRemotionV2; scenePackId: string }> = ({
  plan,
  scenePackId,
}) => {
  const [pack, setPack] = React.useState<any>(null);

  React.useEffect(() => {
    loadScenePack(scenePackId).then(setPack);
  }, [scenePackId]);

  if (!pack) return null;

  return (
    <AbsoluteFill>
      {plan.layers.map((layer) => {
        const base = applyTransform(anchorStyle(layer.anchor), layer);
        const commonWrap: React.CSSProperties = {
          position: "absolute",
          width: "100%",
          height: "100%",
          ...base,
        };

        return (
          <Sequence key={layer.id} from={layer.from} durationInFrames={layer.durationInFrames}>
            <AbsoluteFill style={{ position: "absolute", zIndex: layer.zIndex }}>
              {layer.kind === "VIDEO" && layer.src ? (
                <Video src={layer.src} muted={layer.muted} volume={layer.volume ?? 1} />
              ) : layer.kind === "ALPHA_VIDEO" && layer.alphaSrc ? (
                <div style={commonWrap}>
                  <OffthreadVideo
                    src={layer.alphaSrc.startsWith("http") ? layer.alphaSrc : staticFile(layer.alphaSrc)}
                    transparent
                    muted={layer.muted}
                    volume={layer.volume ?? 0}
                  />
                </div>
              ) : layer.kind === "NATIVE" ? (
                (() => {
                  const Comp = resolveFromPack(pack, layer.componentName);
                  return <Comp beat={layer.props?.beat} variables={layer.props?.variables} />;
                })()
              ) : null}
            </AbsoluteFill>
          </Sequence>
        );
      })}
    </AbsoluteFill>
  );
};

Segment 2/6 — Auto planner: per beat → list of Sora shots (+ overlay preset)

Key concept: a single beat can produce two Sora shots:

background shot (FULL_SCENE or BG_ONLY)

character shot (CHAR_ALPHA) if enabled

// autoShotPlanner.ts
import { StoryIRV1, BeatType } from "./types";
import { ShotType } from "./types.v2";
import { OverlayPreset, OVERLAY_RULES_V1 } from "./formatOverlayRules";
import { SoraShotPolicy, DEFAULT_SORA_SHOT_POLICY } from "./formatShotPolicy";

export type PlannedShot = {
  role: "bg" | "char";
  shotType: ShotType;
  overlayPreset?: OverlayPreset; // only for char
  seconds: number;
  muteOriginalAudio?: boolean;
};

const DEFAULT_PRESETS: OverlayPreset[] = [
  "char_bottom_right",
  "char_bottom_left",
  "char_top_right",
  "char_center",
];

export function planShotsForBeat(opts: {
  ir: StoryIRV1;
  beatIndex: number;
  beat: StoryIRV1["beats"][number];
  policy?: SoraShotPolicy;
}): PlannedShot[] {
  const { beat, beatIndex, policy } = opts;
  const p = { ...DEFAULT_SORA_SHOT_POLICY, ...(policy ?? {}) }[beat.type as BeatType] ?? DEFAULT_SORA_SHOT_POLICY.STEP!;

  const shots: PlannedShot[] = [];

  // Background planning
  if (p.bg === "FULL_SCENE") {
    shots.push({
      role: "bg",
      shotType: "FULL_SCENE",
      seconds: clampSeconds(beat.duration_s),
      muteOriginalAudio: p.audio?.keep === false,
    });
  } else if (p.bg === "BG_ONLY") {
    shots.push({
      role: "bg",
      shotType: "BG_ONLY",
      seconds: clampSeconds(beat.duration_s),
      muteOriginalAudio: true, // bg plates usually silent
    });
  }

  // Character overlay planning
  if (p.charAlpha?.enabled) {
    const rotateEvery = p.charAlpha.rotateEvery ?? 1;
    const presets = p.charAlpha.presets ?? DEFAULT_PRESETS;

    // Only rotate based on beats of the same type (STEPs) typically.
    // For simplicity here: use beatIndex / rotateEvery.
    const slot = Math.floor(beatIndex / rotateEvery) % presets.length;
    const overlayPreset = presets[slot];

    shots.push({
      role: "char",
      shotType: "CHAR_ALPHA",
      overlayPreset,
      seconds: clampSeconds(beat.duration_s),
      muteOriginalAudio: true,
    });
  }

  return shots;
}

function clampSeconds(s: number) {
  return Math.max(1, Math.min(20, Math.round(s)));
}

export function makeShotPlanV3(opts: {
  ir: StoryIRV1;
  format: FormatPackV2;
  policy?: SoraShotPolicy; // override per run if desired
  model?: "sora-2" | "sora-2-pro";
  size?: "720x1280" | "1280x720";
  referenceFileIds?: string[];

  const globalTokens = [

// makeRemotionRenderPlan.v3.ts
import { RenderPlanRemotionV2 } from "./renderPlan.v2";
import { StoryIRV1 } from "./types";
import { AssetManifestV4, indexAssets } from "./assetManifest.v4";
import { FormatPackV2 } from "./types.v2";
import { OVERLAY_RULES_V1 } from "./formatOverlayRules";

export function makeRemotionRenderPlanV3(opts: {
  ir: StoryIRV1;
  format: FormatPackV2 & { overlayRules?: typeof OVERLAY_RULES_V1 };
  assets: AssetManifestV4;
  outSize?: { w: number; h: number };
}): RenderPlanRemotionV2 {
  const { ir, format, assets } = opts;
  const fps = ir.meta.fps;
  const size = opts.outSize ?? (ir.meta.aspect === "9:16" ? { w: 720, h: 1280 } : { w: 1280, h: 720 });

  const overlayRules = format.overlayRules ?? OVERLAY_RULES_V1;
  const { byBeat } = indexAssets(assets);

  let cursor = 0;
  const layers: RenderPlanRemotionV2["layers"] = [];

  for (const beat of ir.beats) {
    const from = cursor;
    const durationInFrames = Math.max(1, Math.round(beat.duration_s * fps));
    const isSoraBeat = format.renderStrategy.soraBeatTypes.includes(beat.type as any);
    const beatClips = byBeat.get(beat.id) ?? [];

    // choose bg clip: prefer FULL_SCENE, else BG_ONLY
    const bg =
      beatClips.find(c => c.role === "bg" && c.shotType === "FULL_SCENE" && c.src) ??
      beatClips.find(c => c.role === "bg" && c.shotType === "BG_ONLY" && c.src);

    if (isSoraBeat && bg?.src) {
      layers.push({
        id: bg_${beat.id},
        kind: "VIDEO",
        zIndex: 0,
        from,
        durationInFrames,
        src: bg.src,
        muted: false,
        volume: 1,
      });

      // optional overlay text (common when bg-only or even full-scene)
      layers.push({
        id: overlay_text_${beat.id},
        kind: "NATIVE",
        zIndex: 30,
        from,
        durationInFrames,
        componentName: "SceneOverlayText",
        props: { beat, variables: ir.variables },
      });
    } else {
      // native scene if not Sora or missing bg
      const componentName = format.componentMap[beat.type] ?? "SceneFallback";
      layers.push({
        id: native_${beat.id},
        kind: "NATIVE",
        zIndex: 0,
        from,
        durationInFrames,
        componentName,
        props: { beat, variables: ir.variables },
      });
    }

    // character overlay if present
    const char = beatClips.find(c => c.role === "char" && c.shotType === "CHAR_ALPHA" && c.alphaSrc);
    if (char?.alphaSrc) {
      const preset = char.overlayPreset ?? overlayRules.defaultPreset;
      const placement = overlayRules.presets[preset];

      layers.push({
        id: char_${beat.id},
        kind: "ALPHA_VIDEO",
        zIndex: 20,
        from,
        durationInFrames,
        alphaSrc: char.alphaSrc,
        transparent: true,
        anchor: placement.anchor,
        transform: placement.transform,
        muted: true,
        volume: 0,
      });
    }

    cursor += durationInFrames;
  }

  layers.sort((a, b) => (a.from - b.from) || (a.zIndex - b.zIndex));
  return { meta: { fps, size }, layers };
}

// validatePipeline.v2.ts
import { StoryIRV1 } from "./types";
import { FormatPackV2, ShotPlanV2 } from "./types.v2";
import { AssetManifestV4 } from "./assetManifest.v4";
import { RenderPlanRemotionV2 } from "./renderPlan.v2";

export function validatePipelineV2(opts: {
  ir: StoryIRV1;
  format: FormatPackV2;
  shotPlan: ShotPlanV2;
  assets: AssetManifestV4;
  renderPlan: RenderPlanRemotionV2;
}) {
  const { ir, shotPlan, assets, renderPlan, format } = opts;

  // fps alignment
  if (ir.meta.fps !== shotPlan.meta.fps) throw new Error("FPS mismatch: IR vs ShotPlan");
  if (ir.meta.fps !== renderPlan.meta.fps) throw new Error("FPS mismatch: IR vs RenderPlan");

  // Each Sora-routed beat must have at least one bg (FULL_SCENE or BG_ONLY)
  const beatIds = ir.beats.map(b => b.id);
  for (const beat of ir.beats) {
    const isSoraBeat = format.renderStrategy.soraBeatTypes.includes(beat.type as any);
    if (!isSoraBeat) continue;

    const beatClips = assets.clips.filter(c => c.beatId === beat.id);
    const hasBg = beatClips.some(c => c.role === "bg" && (c.src && (c.shotType === "FULL_SCENE" || c.shotType === "BG_ONLY")));
    if (!hasBg) throw new Error(Missing BG clip for Sora beat: ${beat.id} (${beat.type}));
  }

  // If a char shot exists, it must have alphaSrc
  for (const c of assets.clips.filter(c => c.role === "char")) {
    if (!c.alphaSrc) throw new Error(CHAR clip missing alphaSrc: shot=${c.shotId} beat=${c.beatId});
  }

  // Render plan sanity
  for (const layer of renderPlan.layers) {
    if (layer.durationInFrames <= 0) throw new Error(Non-positive duration: ${layer.id});
    if (layer.kind === "ALPHA_VIDEO" && !layer.alphaSrc) throw new Error(ALPHA layer missing alphaSrc: ${layer.id});
    if (layer.kind === "VIDEO" && !layer.src) throw new Error(VIDEO layer missing src: ${layer.id});
  }

  return { ok: true };
}

export type ShotBudget = {

  maxTotalSeconds?: number;   // optional cap on sum(seconds) across Sora jobs
  preferKeepHookFullScene?: boolean;
  preferKeepProofFullScene?: boolean;

  // Allow character overlays even when BG is reused
  allowCharAlphaOverSteps: boolean;

  maxSoraJobs: 10,
  maxTotalSeconds: 80,
  preferKeepHookFullScene: true,
  preferKeepProofFullScene: true,
  stepBgPlateCount: 3,
  stepBgReuseMode: "round_robin",
  allowCharAlphaOverSteps: true,
  degradeOrder: ["DROP_CHAR_ALPHA", "DROP_STEP_BG_PLATES", "DOWNGRADE_HOOK_TO_BG_ONLY"],
};

  if (type === "PROOF") return 80;
  if (type === "PROMISE") return 60;

  if (type === "CTA") return 40;

// bgPlatePool.ts
import { StoryIRV1 } from "./types";

export type BgPlate = {
  id: string;            // "plate_step_0"
  promptHint: string;    // e.g. "minimal diagram background"
  seconds: number;       // used once; reused across beats
  intentKey?: string;    // for by_intent matching
};

export function buildStepBgPlatePool(opts: {
  ir: StoryIRV1;
  plateCount: number;
  mode: "round_robin" | "by_intent";
}): BgPlate[] {
  const steps = opts.ir.beats.filter(b => b.type === "STEP");
  if (steps.length === 0) return [];

  if (opts.mode === "by_intent") {
    // cluster by first broll intent key (cheap heuristic)
    const intents = new Map<string, number>();
    for (const s of steps) {
      const key = s.broll?.[0]?.intent ?? "abstract";
      intents.set(key, (intents.get(key) ?? 0) + 1);
    }
    const ranked = [...intents.entries()].sort((a, b) => b[1] - a[1]);
    const chosen = ranked.slice(0, opts.plateCount).map(([k], i) => ({
      id: plate_step_${i},
      intentKey: k,
      promptHint: Generate a clean ${k} style background plate, no text, with empty space for captions.,
      seconds: 4, // keep plates short; loopable or holdable
    }));
    return chosen.length ? chosen : fallbackPool(opts.plateCount);
  }

  return fallbackPool(opts.plateCount);
}

function fallbackPool(n: number): BgPlate[] {
  const hints = [
    "clean abstract tech background plate, no text",
    "minimal diagram grid background plate, no text",
    "simple UI mock background plate, no text",
    "soft gradient flat background plate, no text",
  ];
  return Array.from({ length: n }, (_, i) => ({
    id: plate_step_${i},
    promptHint: hints[i % hints.length],
    seconds: 4,
  }));
}
Segment 3/6 — Budget-aware shot planning (decides what to generate vs reuse)

This outputs a shot request set plus a beat→plate assignment.

// applyBudget.ts
import { StoryIRV1 } from "./types";
import { ShotBudget, DEFAULT_BUDGET, estimateJobSeconds, beatPriority } from "./shotBudget";
import { buildStepBgPlatePool, BgPlate } from "./bgPlatePool";

export type BudgetPlan = {
  // These are the only BG shots we will actually generate
  bgShotsToGenerate: Array<{ key: string; beatId?: string; role: "bg"; type: "HOOK"|"PROOF"|"PROMISE"|"STEP_PLATE"; seconds: number; promptHint: string; }>;

  // Whether we generate char alpha per beat
  charAlphaBeats: Set<string>;

  // For each STEP beat, which plate key to reuse
  stepBeatToPlateKey: Map<string, string>;
};

export function applyShotBudget(ir: StoryIRV1, budget?: Partial<ShotBudget>): BudgetPlan {
  const B: ShotBudget = { ...DEFAULT_BUDGET, ...(budget ?? {}) };

  const stepPlates = buildStepBgPlatePool({ ir, plateCount: B.stepBgPlateCount, mode: B.stepBgReuseMode });

  const bgShotsToGenerate: BudgetPlan["bgShotsToGenerate"] = [];
  const charAlphaBeats = new Set<string>();
  const stepBeatToPlateKey = new Map<string, string>();

  // 1) Decide protected hero BG shots (HOOK/PROOF)
  for (const beat of ir.beats) {
    if (beat.type === "HOOK" && B.preferKeepHookFullScene) {
      bgShotsToGenerate.push({
        key: bg_hook_${beat.id},
        beatId: beat.id,
        role: "bg",
        type: "HOOK",
        seconds: estimateJobSeconds(beat.duration_s),
        promptHint: "FULL_SCENE hook hero shot (character + scene), readable composition.",
      });
    }
    if (beat.type === "PROOF" && B.preferKeepProofFullScene) {
      bgShotsToGenerate.push({
        key: bg_proof_${beat.id},
        beatId: beat.id,
        role: "bg",
        type: "PROOF",
        seconds: estimateJobSeconds(beat.duration_s),
        promptHint: "FULL_SCENE proof/b-roll hero shot, visually explanatory.",
      });
    }
  }

  // 2) Build STEP reusable BG plates
  stepPlates.forEach((p) => {
    bgShotsToGenerate.push({
      key: bg_plate_${p.id},
      role: "bg",
      type: "STEP_PLATE",
      seconds: estimateJobSeconds(p.seconds),
      promptHint: BG_ONLY: ${p.promptHint},
    });
  });

  // 3) Assign each STEP beat to a plate key
  const stepBeats = ir.beats.filter(b => b.type === "STEP");
  stepBeats.forEach((b, i) => {
    const plate = stepPlates[i % Math.max(1, stepPlates.length)];
    stepBeatToPlateKey.set(b.id, bg_plate_${plate.id});
  });

  // 4) Decide char alpha overlays
  if (B.allowCharAlphaOverSteps) {
    for (const beat of ir.beats) {
      if (beat.type === "STEP") charAlphaBeats.add(beat.id);
      if (beat.type === "PROMISE") charAlphaBeats.add(beat.id);
    }
  }

  // 5) Enforce budgets by degrading
  enforceBudgets({ ir, B, bgShotsToGenerate, charAlphaBeats });

  return { bgShotsToGenerate, charAlphaBeats, stepBeatToPlateKey };
}

function enforceBudgets(opts: {
  ir: StoryIRV1;
  B: ShotBudget;
  bgShotsToGenerate: BudgetPlan["bgShotsToGenerate"];
  charAlphaBeats: Set<string>;
}) {
  const { B, bgShotsToGenerate, charAlphaBeats, ir } = opts;

  const countJobs = () => bgShotsToGenerate.length + charAlphaBeats.size;
  const totalSeconds = () =>
    bgShotsToGenerate.reduce((s, x) => s + x.seconds, 0) + Array.from(charAlphaBeats).length * 4; // char shots ~4s baseline

  // degrade until within caps
  while (
    countJobs() > B.maxSoraJobs ||
    (B.maxTotalSeconds !== undefined && totalSeconds() > B.maxTotalSeconds)
  ) {
    const action = B.degradeOrder[0];

    if (action === "DROP_CHAR_ALPHA" && charAlphaBeats.size) {
      // drop lowest priority char alpha first
      const candidates = ir.beats
        .filter(b => charAlphaBeats.has(b.id))
        .sort((a, b) => beatPriority(a.type) - beatPriority(b.type));
      if (candidates.length) charAlphaBeats.delete(candidates[0].id);
      else break;
      continue;
    }

    if (action === "DROP_STEP_BG_PLATES") {
      // remove one STEP plate BG shot (keep at least 1)
      const stepPlates = bgShotsToGenerate.filter(x => x.type === "STEP_PLATE");
      if (stepPlates.length <= 1) {
        // rotate degrade order
        B.degradeOrder.push(B.degradeOrder.shift()!);
        continue;
      }
      bgShotsToGenerate.splice(bgShotsToGenerate.indexOf(stepPlates[stepPlates.length - 1]), 1);
      continue;
    }

    if (action === "DOWNGRADE_HOOK_TO_BG_ONLY") {
      const hookShots = bgShotsToGenerate.filter(x => x.type === "HOOK");
      if (!hookShots.length) {
        B.degradeOrder.push(B.degradeOrder.shift()!);
        continue;
      }
      // just shorten it instead of removing
      hookShots[0].seconds = Math.max(2, Math.floor(hookShots[0].seconds / 2));
      continue;
    }

    // rotate degrade order if current action can't help
    B.degradeOrder.push(B.degradeOrder.shift()!);
  }
}

// makeShotPlan.budgeted.ts
import crypto from "node:crypto";
import { StoryIRV1 } from "./types";
import { ShotPlanV2 } from "./types.v2";
import { BudgetPlan } from "./applyBudget";
import { buildSoraPrompt } from "./soraShotTypes";

function sha256(s: string) {
  return crypto.createHash("sha256").update(s).digest("hex");
}

export function makeShotPlanBudgeted(opts: {
  ir: StoryIRV1;
  budgetPlan: BudgetPlan;
  model?: "sora-2" | "sora-2-pro";
  size?: "720x1280" | "1280x720";
  referenceFileIds?: string[];
}): ShotPlanV2 & { reuseMap: Record<string, string> } {
  const { ir, budgetPlan } = opts;
  const model = opts.model ?? "sora-2";
  const size = opts.size ?? (ir.meta.aspect === "9:16" ? "720x1280" : "1280x720");

  const globalTokens = ["clean flat 2D explainer", "high contrast", "safe margins"];
  const negativeTokens = ["tiny text", "watermarks", "busy background"];

  // reuseKey -> shotId (only these get generated)
  const reuseMap: Record<string, string> = {};

  const shots: any[] = [];

  // 1) BG shots to generate (including reusable STEP plates)
  for (const bg of budgetPlan.bgShotsToGenerate) {
    const prompt =
      bg.type === "STEP_PLATE"
        ? buildSoraPrompt({ shotType: "BG_ONLY", beatNarration: bg.promptHint, globalTokens, negativeTokens })
        : buildSoraPrompt({ shotType: "FULL_SCENE", beatNarration: bg.promptHint, globalTokens, negativeTokens });

    const cacheKey = sha256(JSON.stringify({ model, size, prompt, refs: opts.referenceFileIds ?? [] }));

    const shotId = shot_${bg.key};

Segment 2/6 — Decide how a plate covers a beat (duration matcher)

// matchPlateToBeat.ts
import { BgFillMode, PlateVariantPlan } from "./fillModes";

export function matchPlateToBeat(opts: {
  beatFrames: number;
  plateFrames: number;
  // how much randomness you want in plate “entry point”
  maxTrimOffsetFrames?: number;
  preferStretchOverLoop?: boolean;
}): PlateVariantPlan {
  const { beatFrames, plateFrames } = opts;
  const maxTrim = Math.max(0, opts.maxTrimOffsetFrames ?? Math.floor(plateFrames * 0.5));

  const trimOffsetFrames =
    maxTrim === 0 ? 0 : Math.floor(Math.random() * (maxTrim + 1));

  // If beat is shorter than the plate, just trim into the plate and let it run.
  if (beatFrames <= plateFrames) {
    return { fillMode: "HOLD_LAST", trimOffsetFrames };
  }

  // Beat longer than plate: either stretch or loop.
  if (opts.preferStretchOverLoop) {
    // playbackRate < 1 slows it down (longer), > 1 speeds it up (shorter)
    const playbackRate = plateFrames / beatFrames;
    return { fillMode: "STRETCH", trimOffsetFrames, playbackRate };
  }

  return { fillMode: "LOOP", trimOffsetFrames };
}

// varietyInjector.ts
import { StoryIRV1 } from "./types";
import { ShotBudget } from "./shotBudget";
import { BudgetPlan } from "./applyBudget";

function intentKey(beat: StoryIRV1["beats"][number]) {
  return beat.broll?.[0]?.intent ?? beat.type ?? "unknown";
}

export function injectVariety(opts: {
  ir: StoryIRV1;
  budget: ShotBudget;
  plan: BudgetPlan;
}): BudgetPlan {
  const { ir, budget, plan } = opts;

  // headroom = how many more Sora jobs we can add (rough)
  const currentJobs = plan.bgShotsToGenerate.length + plan.charAlphaBeats.size;
  let remaining = Math.max(0, budget.maxSoraJobs - currentJobs);
  if (remaining <= 0) return plan;

  // Find “shift points” in STEP beats where intent changes
  const steps = ir.beats.filter((b) => b.type === "STEP");
  const shifts: Array<{ beatId: string; key: string }> = [];
  for (let i = 1; i < steps.length; i++) {
    const prev = intentKey(steps[i - 1]);
    const cur = intentKey(steps[i]);
    if (prev !== cur) shifts.push({ beatId: steps[i].id, key: cur });
  }

  // Spend up to N extra plates (where N=remaining)
  for (let i = 0; i < shifts.length && remaining > 0; i++) {
    const s = shifts[i];
    const plateId = plate_step_extra_${i};
    const plateKey = bg_plate_${plateId};

    plan.bgShotsToGenerate.push({
      key: plateKey,
      role: "bg",
      type: "STEP_PLATE",
      seconds: 4,
      promptHint: BG_ONLY: Generate a clean ${s.key} themed background plate, no text, safe margins for captions.,
    });

    // Reassign this beat (and optionally the next 1–2 beats) to use the new plate
    plan.stepBeatToPlateKey.set(s.beatId, plateKey);
    remaining--;
  }

  return plan;
}

Segment 2/6 — Detect plate anti-patterns across your STEP beats

// detectPlateAntiPatterns.ts
import { DEFAULT_RISK_CONFIG, PlateUsage, RiskReport } from "./antiPatternTypes";

function isExactMultiple(a: number, b: number) {
  return b > 0 && a % b === 0;
}

export function detectPlateAntiPatterns(usages: PlateUsage[], cfg = DEFAULT_RISK_CONFIG): RiskReport[] {
  const reports: RiskReport[] = [];

  let samePlateStreak = 1;
  let sameTrimStreak = 1;

  for (let i = 0; i < usages.length; i++) {
    const u = usages[i];
    const prev = usages[i - 1];

    if (prev && prev.plateKey === u.plateKey) samePlateStreak++;
    else samePlateStreak = 1;

    const prevTrim = prev?.variant.trimOffsetFrames ?? 0;
    const curTrim = u.variant.trimOffsetFrames ?? 0;
    if (prev && prevTrim === curTrim) sameTrimStreak++;
    else sameTrimStreak = 1;

    const seamRepeat = (u.variant.fillMode === "LOOP") && isExactMultiple(u.beatFrames, u.plateFrames);

    const playbackRate = u.variant.playbackRate ?? 1;
    const extremeStretch =
      u.variant.fillMode === "STRETCH" &&
      (playbackRate < cfg.minPlaybackRate || playbackRate > cfg.maxPlaybackRate);

    const samePlateRun = samePlateStreak > cfg.maxSamePlateConsecutive;
    const sameTrimRun = sameTrimStreak > cfg.maxSameTrimConsecutive;

    let score = 0;
    if (seamRepeat) score += cfg.seamRepeatPenalty;
    if (samePlateRun) score += cfg.samePlatePenalty;
    if (sameTrimRun) score += cfg.sameTrimPenalty;
    if (extremeStretch) score += cfg.extremeStretchPenalty;

    reports.push({
      usage: u,
      score: Math.min(100, score),
      flags: { seamRepeat, samePlateRun, sameTrimRun, extremeStretch },
    });
  }

  return reports;
}
Segment 3/6 — Auto-fix: switch LOOP→STRETCH, randomize trim, or HOLD_LAST

// fixPlateVariants.ts
import { PlateUsage, RiskReport, DEFAULT_RISK_CONFIG } from "./antiPatternTypes";
import { PlateVariantPlan } from "./fillModes";

function clamp(n: number, a: number, b: number) {
  return Math.max(a, Math.min(b, n));
}

function randInt(maxInclusive: number) {
  return Math.floor(Math.random() * (maxInclusive + 1));
}

export function autoFixPlateVariants(opts: {
  usages: PlateUsage[];
  reports: RiskReport[];
  cfg?: typeof DEFAULT_RISK_CONFIG;
}): PlateUsage[] {
  const cfg = opts.cfg ?? DEFAULT_RISK_CONFIG;

  return opts.usages.map((u, idx) => {
    const r = opts.reports[idx];
    if (!r || r.score < 25) return u;

    const v: PlateVariantPlan = { ...(u.variant ?? {}) };
    const maxTrim = Math.floor(u.plateFrames * 0.45);

    // 1) If LOOP seam repeats, prefer STRETCH (within sane playbackRate bounds)
    if (r.flags.seamRepeat && u.plateFrames > 0) {
      const rawRate = u.plateFrames / u.beatFrames; // <1 slows down
      v.fillMode = "STRETCH";
      v.playbackRate = clamp(rawRate, cfg.minPlaybackRate, cfg.maxPlaybackRate);
    }

    // 2) If same trim repeats, re-roll trim
    if (r.flags.sameTrimRun) {
      v.trimOffsetFrames = randInt(Math.max(0, maxTrim));
    } else if (v.trimOffsetFrames === undefined) {
      v.trimOffsetFrames = randInt(Math.max(0, maxTrim));
    }

    // 3) If stretch is extreme, fall back to LOOP (but with randomized trim)
    if (r.flags.extremeStretch) {
      v.fillMode = "LOOP";
      v.playbackRate = undefined;
      v.trimOffsetFrames = randInt(Math.max(0, maxTrim));
    }

    // 4) If beat shorter than plate, HOLD_LAST is super safe
    if (u.beatFrames <= u.plateFrames) {
      v.fillMode = "HOLD_LAST"; // last frame stays visible by default :contentReference[oaicite:3]{index=3}
      v.playbackRate = undefined;
    }

    return { ...u, variant: v };
  });
}

Segment 5/6 — Bonus: detect “character overlay boredom” and rotate presets faster

// charBoredom.ts
import { OverlayPreset } from "./formatOverlayRules";

export type CharPlacement = { beatId: string; preset: OverlayPreset };

export function fixCharPlacementBoredom(opts: {
  placements: CharPlacement[];
  maxSamePresetConsecutive?: number; // e.g. 2
  presetsCycle: OverlayPreset[];
}): CharPlacement[] {
  const maxSame = opts.maxSamePresetConsecutive ?? 2;
  const out: CharPlacement[] = [];

  let streak = 0;
  let last: OverlayPreset | null = null;
  let cycleIdx = 0;

  for (const p of opts.placements) {
    if (p.preset === last) streak++;
    else streak = 1;

    let preset = p.preset;

    if (streak > maxSame) {
      // advance cycle until it differs
      let tries = 0;
      while (tries < opts.presetsCycle.length && opts.presetsCycle[cycleIdx] === last) {
        cycleIdx = (cycleIdx + 1) % opts.presetsCycle.length;
        tries++;
      }
      preset = opts.presetsCycle[cycleIdx];
      cycleIdx = (cycleIdx + 1) % opts.presetsCycle.length;
      streak = 1;
    }

    out.push({ beatId: p.beatId, preset });
    last = preset;
  }

  return out;
}

  frame: number;
  cues: NarrationCue[];
  policy: DuckingPolicy;
}): number {
  const { frame, cues, policy } = opts;
  if (!policy.enabled) return policy.bgBaseVolume;

  const inCue = cues.some((c) => frame >= c.from && frame < c.from + c.durationInFrames);

  const sr = opts.sampleRate ?? 48000;

  const args = [

    "-f", "lavfi",
    "-i", anullsrc=r=${sr}:cl=mono,
    "-t", ${seconds},
    opts.outPath,

}

  // Simple loudness normalization. You can tune target if desired.

    "-y",
    "-i", opts.inPath,
    "-af", "loudnorm=I=-16:TP=-1.5:LRA=11",

  // concat demuxer

    "-y",
    "-f", "concat",
    "-safe", "0",
    "-i", listFile,
    "-c", "copy",
    opts.outPath,

Segment 4/6 — Generate per-beat WAVs with optional silence padding

// synthesizeBeats.ts
import path from "node:path";
import { TTSProvider } from "./ttsProvider";
import { BeatNarrationInput, NarrationAsset } from "./narrationTypes";
import { generateSilenceWav, normalizeLoudnessWav, concatWavs, probeAudioDurationSeconds } from "./audioUtils";

export async function synthesizeBeatNarrations(opts: {
  provider: TTSProvider;
  beats: BeatNarrationInput[];
  outDir: string;
  voiceId?: string;
}): Promise<NarrationAsset[]> {
  const assets: NarrationAsset[] = [];

  for (const b of opts.beats) {
    const rawPath = path.join(opts.outDir, vo_${b.beatId}_raw.wav);
    const normPath = path.join(opts.outDir, vo_${b.beatId}.wav);

    await opts.provider.synthesize({
      text: b.text,
      outPath: rawPath,
      voiceId: opts.voiceId,
    });

    await normalizeLoudnessWav({ inPath: rawPath, outPath: normPath });

    // Add optional pre/post silence by concatenating [pre][voice][post]
    const parts: string[] = [];
    if ((b.preSilenceMs ?? 0) > 0) {
      const pre = path.join(opts.outDir, sil_pre_${b.beatId}.wav);
      await generateSilenceWav({ outPath: pre, ms: b.preSilenceMs! });
      parts.push(pre);
    }

    parts.push(normPath);

    if ((b.postSilenceMs ?? 0) > 0) {
      const post = path.join(opts.outDir, sil_post_${b.beatId}.wav);
      await generateSilenceWav({ outPath: post, ms: b.postSilenceMs! });
      parts.push(post);
    }

    let finalPath = normPath;
    if (parts.length > 1) {
      finalPath = path.join(opts.outDir, vo_${b.beatId}_padded.wav);
      await concatWavs({ inputPaths: parts, outPath: finalPath });
    }

    const dur = await probeAudioDurationSeconds(finalPath);
    assets.push({ beatId: b.beatId, wavPath: finalPath, durationSeconds: dur });
  }

  return assets;
}

import path from "node:path";
import { NarrationAsset, StitchedNarration, NarrationCues } from "./narrationTypes";
import { concatWavs, wavToMp3 } from "./audioUtils";

export function secondsToFrames(sec: number, fps: number) {
  return Math.max(1, Math.round(sec * fps));
}

export async function stitchNarration(opts: {
  assets: NarrationAsset[];
  outDir: string;
  fps: number;
  alsoMp3?: boolean;
}): Promise<StitchedNarration> {
  const stitchedWavPath = path.join(opts.outDir, "narration_stitched.wav");
  await concatWavs({ inputPaths: opts.assets.map(a => a.wavPath), outPath: stitchedWavPath });

  const cues: NarrationCues = {};
  let cursorSeconds = 0;

  for (const a of opts.assets) {
    const startSeconds = cursorSeconds;
    const durationSeconds = a.durationSeconds;

    const fromFrame = Math.round(startSeconds * opts.fps);
    const durationInFrames = secondsToFrames(durationSeconds, opts.fps);

    cues[a.beatId] = { fromFrame, durationInFrames, startSeconds, durationSeconds };

    cursorSeconds += durationSeconds;
  }

  let stitchedMp3Path: string | undefined;
  if (opts.alsoMp3) {
    stitchedMp3Path = path.join(opts.outDir, "narration_stitched.mp3");
    await wavToMp3({ inPath: stitchedWavPath, outPath: stitchedMp3Path });
  }

  return { stitchedWavPath, stitchedMp3Path, totalSeconds: cursorSeconds, cues };
}

Segment 1/6 — Speech timing model (estimate + safety buffers)

// speechTiming.ts
export type SpeechTimingConfig = {
  targetWpm: number;          // 150–175 is common; 140 for “calm/clear”
  minBeatSeconds: number;     // never shorter than this
  leadInSeconds: number;      // before speaking starts (breath room)
  tailSeconds: number;        // after speaking ends (don’t hard cut)
  punctuationPauseBoost: number; // extra seconds added based on punctuation count
};

export const DEFAULT_SPEECH_TIMING: SpeechTimingConfig = {
  targetWpm: 155,
  minBeatSeconds: 1.2,
  leadInSeconds: 0.10,
  tailSeconds: 0.18,
  punctuationPauseBoost: 0.015, // tiny extra per punctuation mark
};

export function estimateSpeechSeconds(text: string, cfg = DEFAULT_SPEECH_TIMING): number {
  const words = (text.trim().match(/\S+/g) ?? []).length;
  const base = (words / cfg.targetWpm) * 60;

  // crude pause proxy
  const punct = (text.match(/[,\.\!\?\:\;—-]/g) ?? []).length;
  const pause = punct * cfg.punctuationPauseBoost;

  const total = base + pause + cfg.leadInSeconds + cfg.tailSeconds;
  return Math.max(cfg.minBeatSeconds, round2(total));
}

function round2(n: number) {
  return Math.round(n * 100) / 100;
}
Segment 2/6 — Reconcile beat durations from real TTS durations (preferred)

// reconcileBeats.ts
import { StoryIRV1 } from "./types";
import { StitchedNarration } from "./narrationTypes";
import { SpeechTimingConfig, DEFAULT_SPEECH_TIMING, estimateSpeechSeconds } from "./speechTiming";

export type ReconcileResult = {
  ir: StoryIRV1;
  changes: Array<{ beatId: string; old_s: number; new_s: number; reason: string }>;
  totalOld_s: number;
  totalNew_s: number;
};

export function reconcileBeatDurations(opts: {
  ir: StoryIRV1;
  narration?: StitchedNarration;        // if present, uses real durations per beat
  cfg?: Partial<SpeechTimingConfig>;
}): ReconcileResult {
  const cfg: SpeechTimingConfig = { ...DEFAULT_SPEECH_TIMING, ...(opts.cfg ?? {}) };

  const changes: ReconcileResult["changes"] = [];
  const totalOld_s = sum(opts.ir.beats.map(b => b.duration_s));

  const beats = opts.ir.beats.map((b) => {
    const old = b.duration_s;

    // Preferred: exact beat narration duration from stitched cues
    const cueDur = opts.narration?.cues?.[b.id]?.durationSeconds;

    // Fallback: estimate from text
    const est = estimateSpeechSeconds(b.narration ?? "", cfg);

    const needed = (cueDur ?? est);

    // Don’t force beats shorter than the speech window
    const next = Math.max(old, needed);

    if (Math.abs(next - old) > 0.01) {
      changes.push({
        beatId: b.id,
        old_s: old,
        new_s: next,
        reason: cueDur ? "TTS duration + buffers" : "WPM estimate + buffers",
      });
    }

    return { ...b, duration_s: next };
  });

  const totalNew_s = sum(beats.map(b => b.duration_s));
  return { ir: { ...opts.ir, beats }, changes, totalOld_s, totalNew_s };
}

function sum(xs: number[]) {
  return Math.round(xs.reduce((a, b) => a + b, 0) * 100) / 100;
}
Segment 3/6 — If total runtime exceeds budget: compress safely (without rushing)

Order of operations (gentle → aggressive):

reduce tail/lead buffers slightly

speed up VO a little (e.g. max 1.08–1.12)

split long beats into two beats (best for clarity)

// runtimeBudget.ts
export type RuntimeBudget = {
  maxTotalSeconds: number;
  maxVOSpeedup: number; // e.g. 1.10 (10% faster max)
};

export const DEFAULT_RUNTIME_BUDGET: RuntimeBudget = {
  maxTotalSeconds: 60,
  maxVOSpeedup: 1.10,
};

export type CompressionPlan = {
  voSpeedup?: number; // 1.0..maxVOSpeedup
  bufferScale?: number; // 0.6..1.0
  note: string;
};

export function computeCompressionPlan(totalNew_s: number, budget = DEFAULT_RUNTIME_BUDGET): CompressionPlan | null {
  if (totalNew_s <= budget.maxTotalSeconds) return null;

  const over = totalNew_s - budget.maxTotalSeconds;
  const ratioNeeded = budget.maxTotalSeconds / totalNew_s; // <1

  // try buffer reduction first (assume ~10% is buffer-ish across a video)
  const bufferScale = Math.max(0.6, Math.min(1.0, ratioNeeded / 0.95));
  const afterBuffer = totalNew_s * (0.95 * bufferScale + 0.05);

  if (afterBuffer <= budget.maxTotalSeconds) {
    return { bufferScale, note: Reduced buffers to fit budget (over by ${over.toFixed(2)}s). };
  }

  // then VO speedup (small)
  const voSpeedup = Math.min(budget.maxVOSpeedup, totalNew_s / budget.maxTotalSeconds);
  return {
    bufferScale,
    voSpeedup,
    note: Reduced buffers + sped VO up to ${(voSpeedup * 100 - 100).toFixed(1)}%.,
  };
}

  if (opts.speed <= 0.5 || opts.speed >= 2.0) throw new Error("speed must be between 0.5 and 2.0");

  const args = [
    "-y",
    "-i", opts.inPath,
    "-filter:a", atempo=${opts.speed.toFixed(4)},
    opts.outPath,
  ];
  await execFileAsync("ffmpeg", args);
}
After speeding up, you should re-probe duration and rebuild cues (or scale cue durations by 1/speed if you keep per-beat WAVs).

Segment 5/6 — Propagate reconciled durations into Sora shot seconds + Remotion frames

This guarantees the actor/video isn’t cut off.

// propagateDurations.ts
import { StoryIRV1 } from "./types";
import { ShotPlanV2 } from "./types.v2";

export function updateShotPlanSecondsFromIR(opts: {
  ir: StoryIRV1;
  shotPlan: ShotPlanV2 & { shots: any[] };
  // optional clamp so you don’t generate huge Sora clips
  clamp?: { min: number; max: number };
}) {
  const clamp = opts.clamp ?? { min: 2, max: 20 };
  const beatById = new Map(opts.ir.beats.map(b => [b.id, b]));

  for (const s of opts.shotPlan.shots) {
    const beat = beatById.get(s.fromBeatId);
    if (!beat) continue;

    // Make sure any “bg” or “char” shots cover the beat duration.
    const desired = beat.duration_s;
    s.seconds = Math.max(clamp.min, Math.min(clamp.max, Math.round(desired)));

    // If you’re using reusable STEP plates, keep them short (e.g. 4s) and rely on looping:
    if (String(s.id).includes("plate_step")) {
      s.seconds = 4;
    }
  }

  return opts.shotPlan;
}

export function secondsToFrames(sec: number, fps: number) {
  return Math.max(1, Math.round(sec * fps));
}

export async function orchestratePacingSafe(opts: any) {
  // 1) Create IR
  let ir = makeStoryIR(opts.trend, opts.brief);

  // 2) Generate per-beat narration WAVs (so we know real durations)
  const beatInputs = ir.beats.map((b: any) => ({
    beatId: b.id,
    text: b.narration,
    preSilenceMs: 100,
    postSilenceMs: 140,
  }));

  const narrationAssets = await synthesizeBeatNarrations({
    provider: opts.ttsProvider,
    beats: beatInputs,
    outDir: opts.audioOutDir,
    voiceId: opts.voiceId,
  });

  let stitched = await stitchNarration({
    assets: narrationAssets,
    outDir: opts.audioOutDir,
    fps: ir.meta.fps,
    alsoMp3: true,
  });

  // 3) Reconcile beat durations to REAL narration (prevents rushing/cutoffs)
  const recon = reconcileBeatDurations({ ir, narration: stitched });
  ir = recon.ir;

  // 4) If total exceeds runtime budget, compress gently (buffers + small VO speedup)
  const compress = computeCompressionPlan(recon.totalNew_s, opts.runtimeBudget);
  if (compress?.voSpeedup && compress.voSpeedup > 1.0001) {
    const sped = path.join(opts.audioOutDir, "narration_stitched_sped.wav");
    await speedUpAudio({ inPath: stitched.stitchedWavPath, outPath: sped, speed: compress.voSpeedup });

    // simplest: re-stitch cues by scaling durations (best: re-probe each beat audio if you keep per-beat WAVs)
    // For now: rebuild stitched using scaled durations:
    const scaledAssets = narrationAssets.map(a => ({
      ...a,
      durationSeconds: a.durationSeconds / compress.voSpeedup!,
    }));
    stitched = await stitchNarration({ assets: scaledAssets, outDir: opts.audioOutDir, fps: ir.meta.fps, alsoMp3: true });
    stitched.stitchedWavPath = sped;
  }

  // 5) Build shot plan (budgeted or not), then update shot seconds to match beat durations
  let shotPlan = opts.makeShotPlan({ ir, budget: opts.budget }); // your existing function
  shotPlan = updateShotPlanSecondsFromIR({ ir, shotPlan });

  // 6) Run Sora (shots now long enough), build assets, then render plan
  const assets = await opts.runSora({ shotPlan });
  const renderPlan = opts.makeRenderPlan({ ir, assets, narration: stitched });

  return { ir, stitchedNarration: stitched, shotPlan, assets, renderPlan };
}

// hfTtsProvider.ts
import fs from "node:fs/promises";

export type HFTTSConfig = {
  hfToken: string;
  modelId: string;     // e.g. "org/model"
};

export async function hfTtsSynthesize(opts: {
  cfg: HFTTSConfig;
  text: string;
  outPath: string;     // write audio bytes here (wav/mp3 depending on model)
}): Promise<void> {
  const url = https://api-inference.huggingface.co/models/${opts.cfg.modelId};

  const res = await fetch(url, {
    method: "POST",
    headers: {
      Authorization: Bearer ${opts.cfg.hfToken},
      "Content-Type": "application/json",
    },
    body: JSON.stringify({ inputs: opts.text }),
  });

  if (!res.ok) {
    const msg = await res.text().catch(() => "");
    throw new Error(HF TTS failed (${res.status}): ${msg});
  }

  const buf = Buffer.from(await res.arrayBuffer());
  await fs.writeFile(opts.outPath, buf);
}
If your chosen HF model returns something other than raw audio bytes, you adapt the parsing here once, and everything upstream stays the same.

        name: "hf",

  const renderPlan = opts.makeRenderPlan({ ir, assets, voice, stitchedNarration });

  return { voice, ir, stitchedNarration, shotPlan, assets, renderPlan };

import { VoiceEngine } from "./voiceEngine";
import { reconcileBeatDurations } from "./reconcileBeats";

function noTalkingPolicy(ir: any, strategy: VoiceStrategyV3) {
  const beat: any = {};
  for (const b of ir.beats) {
    beat[b.id] = {
      muteSoraAudio: strategy.constraints.ambienceOnlyWhenNarrated,
      forbidTalkingVisuals: strategy.constraints.forbidOnScreenTalkingWhenNarrated,
      soraDialogue: { enabled: false, maxSeconds: 0 },
    };
  }
  return { beat };
}

export const ExternalNarratorEngine: VoiceEngine = {
  name: "external_narrator_hf",
  async build({ ir, strategy, fps, outDir, env }) {
    if (!strategy.narrator) throw new Error("Missing narrator config for EXTERNAL_NARRATOR");

    const inputs = ir.beats.map((b: any) => ({
      beatId: b.id,
      text: b.narration,        // ensure your script generator outputs 3rd person
      preSilenceMs: 100,
      postSilenceMs: 140,
    }));

    const assets = await synthesizeBeatNarrations({
      provider: {
        name: "hf",
        async synthesize(req: any) {
          await hfTtsSynthesize({
            cfg: { hfToken: env.hfToken!, modelId: strategy.narrator!.modelId },
            text: req.text,
            outPath: req.outPath,
          });
        },
      },
      beats: inputs,
      outDir,
    });

    const stitched = await stitchNarration({ assets, outDir, fps, alsoMp3: false });

    // reconcile beat durations to real VO (prevents rushing/cutoff)
    const recon = reconcileBeatDurations({ ir, narration: stitched });
    ir = recon.ir;

    return {
      ir,
      audioPlan: {
        kind: "STITCHED_NARRATION",
        stitchedWavPath: stitched.stitchedWavPath,
        cues: Object.fromEntries(
          Object.entries(stitched.cues).map(([beatId, c]) => [
            beatId,
            { fromFrame: c.fromFrame, durationInFrames: c.durationInFrames, durationSeconds: c.durationSeconds },
          ])
        ),
        ducking: { bgBase: 0.9, ducked: 0.18, fadeFrames: 6 },
      },
      shotAudioPolicy: noTalkingPolicy(ir, strategy),
    };
  },
};

export const SoraDialogueEngine: VoiceEngine = {
  name: "sora_dialogue",
  async build({ ir, strategy }) {
    const allow = new Set(strategy.soraDialogue?.allowBeatTypes ?? []);
    const beat: any = {};

    for (const b of ir.beats) {
      const isDialogue = allow.has(b.type);
      beat[b.id] = {
        muteSoraAudio: !(strategy.soraDialogue?.keepSoraAudio ?? true) || !isDialogue,
        forbidTalkingVisuals: false,
        soraDialogue: {
          enabled: isDialogue,
          maxSeconds: strategy.soraDialogue?.maxSecondsPerBeat ?? 4,
        },
      };

      // optional: cap dialogue beats so they stay snappy
      if (isDialogue) b.duration_s = Math.min(b.duration_s, beat[b.id].soraDialogue.maxSeconds);
    }

    return { ir, audioPlan: { kind: "NONE" }, shotAudioPolicy: { beat } };
  },

Segment 5/6 — Use shotAudioPolicy to modify Sora prompts + shot seconds (no branching elsewhere)

// applyVoiceToShotPlan.ts
import { ShotPlanV2 } from "./types.v2";
import { ShotAudioPolicy } from "./voiceEngine";

const NO_TALKING_TOKENS = [
  "no visible speaking",
  "no lip movement",
  "no dialogue text on screen",
  "mouth closed or neutral expression",
];

const AMBIENCE_ONLY_TOKENS = ["no speech audio", "ambient sound only", "no voiceover"];

export function applyVoicePolicyToShotPlan(opts: {
  shotPlan: ShotPlanV2 & { shots: any[] };
  policy: ShotAudioPolicy;
}) {
  for (const s of opts.shotPlan.shots) {
    const p = opts.policy.beat[s.fromBeatId];
    if (!p) continue;

    // audio
    s.postprocess = s.postprocess ?? {};
    s.postprocess.muteOriginalAudio = p.muteSoraAudio;

    // prompt tokens
    s.extraPromptTokens = s.extraPromptTokens ?? [];
    if (p.forbidTalkingVisuals) s.extraPromptTokens.push(...NO_TALKING_TOKENS);
    if (p.muteSoraAudio) s.extraPromptTokens.push(...AMBIENCE_ONLY_TOKENS);

    // if this beat is Sora dialogue, ensure seconds cover it (or cap)

This is deliberately simple and safe (no web calls, no big NLP). It catches obvious “I / you” cases.

// enforcePerspective.ts
import { NarrationPerspective } from "./voiceVars";

export function enforcePerspective(opts: {
  text: string;
  perspective: NarrationPerspective;
  mode: "STRICT" | "SOFT_REWRITE";
}): { text: string; changed: boolean; warnings: string[] } {
  const warnings: string[] = [];
  let t = opts.text;

  if (opts.perspective !== "third_person") {
    return { text: t, changed: false, warnings };
  }

  const hasFirst = /\b(I|I'm|I’ve|I've|me|my|mine|we|our|ours)\b/i.test(t);
  const hasSecond = /\b(you|you're|you’ve|you've|your|yours)\b/i.test(t);

  if (!hasFirst && !hasSecond) return { text: t, changed: false, warnings };

  if (opts.mode === "STRICT") {
    warnings.push("Narration contains first/second-person language but third-person was required.");
    return { text: t, changed: false, warnings };
  }

  // SOFT_REWRITE: basic conversions
  // NOTE: this is intentionally conservative; for best results, generate 3rd-person in script step.
  t = t
    .replace(/\bI am\b/gi, "He is")
    .replace(/\bI'm\b/gi, "He is")
    .replace(/\bI\b/gi, "He")
    .replace(/\bme\b/gi, "him")
    .replace(/\bmy\b/gi, "his")
    .replace(/\bwe\b/gi, "they")
    .replace(/\bour\b/gi, "their")
    .replace(/\byou are\b/gi, "they are")
    .replace(/\byou\b/gi, "they")
    .replace(/\byour\b/gi, "their");

  warnings.push("Rewrote narration to approximate third-person perspective.");
  return { text: t, changed: true, warnings };
}
If you prefer “the narrator” instead of “he/they”, swap the replacements to “the narrator / Isaiah / the creator”.

// voiceValidation.heuristics.ts
export function hasFirstOrSecondPerson(text: string): boolean {
  if (!text) return false;
  const first = /\b(I|I'm|I’ve|I've|me|my|mine|we|our|ours)\b/i.test(text);
  const second = /\b(you|you're|you’ve|you've|your|yours)\b/i.test(text);
  return first || second;
}

// If your IR marks dialogue beats explicitly, use that instead.
// Fallback: treat beats with "dialogue" field or quotation marks as dialogue intent.
export function beatWantsDialogue(beat: any): boolean {
  if (beat.dialogue && String(beat.dialogue).trim().length > 0) return true;
  const t = (beat.narration ?? "") as string;
  return /["“”]/.test(t) || /\b(says|said|asks|asked|replies|shouts)\b/i.test(t);
}

Segment 4/6 — Validate per-beat conflicts (dialogue vs narration + perspective)

// validateBeats.ts
import { VoiceValidationContext, VoiceValidationIssue } from "./voiceValidation.types";
import { hasFirstOrSecondPerson, beatWantsDialogue } from "./voiceValidation.heuristics";

export function validateBeats(ctx: VoiceValidationContext): VoiceValidationIssue[] {
  const issues: VoiceValidationIssue[] = [];
  const allowDialogue = new Set(ctx.allowDialogueBeatTypes ?? []);

  for (const b of ctx.ir.beats as any[]) {
    const wantsDialogue = beatWantsDialogue(b);
    const narrationText = (b.narration ?? "") as string;

    // Third-person enforcement
    if (ctx.narrationEnabled && ctx.voiceVars.useThirdPersonTTS && ctx.voiceVars.enforcePerspective === "STRICT") {
      if (hasFirstOrSecondPerson(narrationText)) {
        issues.push({
          code: "PERSPECTIVE_VIOLATION",
          level: "ERROR",
          beatId: b.id,
          message: "Narration contains first/second-person language but STRICT third-person was required.",
        });
      }
    }

    // Dialogue beat rules
    if (ctx.mode === "EXTERNAL_NARRATOR") {
      // If the beat is trying to be dialogue but narration-only format, block it
      if (wantsDialogue) {
        issues.push({
          code: "DIALOGUE_NOT_ALLOWED",
          level: "ERROR",
          beatId: b.id,
          message: "Beat appears to be dialogue, but voice mode is EXTERNAL_NARRATOR (no Sora dialogue allowed).",
        });
      }
    }

    if (ctx.mode === "SORA_DIALOGUE") {
      // If Sora dialogue mode: narration should be empty or treated as stage directions only
      if (narrationText.trim().length > 0 && ctx.voiceVars.useThirdPersonTTS) {
        issues.push({
          code: "NARRATION_PRESENT_IN_DIALOGUE_MODE",
          level: "WARN",
          beatId: b.id,
          message: "Narration text is present but mode is SORA_DIALOGUE. Consider moving this into dialogue/stage directions.",
        });
      }
    }

    if (ctx.mode === "HYBRID") {
      const beatTypeAllowed = allowDialogue.has(b.type);

      if (wantsDialogue && !beatTypeAllowed) {
        issues.push({
          code: "DIALOGUE_BEATTYPE_BLOCKED",
          level: "ERROR",
          beatId: b.id,
          message: Beat looks like dialogue but beat.type="${b.type}" is not allowed for Sora dialogue in this format.,
        });
      }

      // Prevent "double voice" unless explicitly allowed
      const isDialogueBeat = wantsDialogue && beatTypeAllowed;
      if (isDialogueBeat && ctx.narrationEnabled && !ctx.allowNarrationAndDialogueSameBeat) {
        issues.push({
          code: "DOUBLE_VOICE_CONFLICT",
          level: "ERROR",
          beatId: b.id,
          message: "Beat is marked for Sora dialogue but narration is also enabled. Disable narration on this beat or allow overlap explicitly.",
        });
      }
    }
  }

  return issues;
}

  estimateWpm: number;

  maxUtilization: { EXTERNAL_NARRATOR: 0.78, SORA_DIALOGUE: 0.62 },
  buffers: { leadIn_s: 0.10, tail_s: 0.18 },
  clipLimits: { maxDialogueClipSeconds: 6, maxNarratedBeatSeconds: 18 },
  estimateWpm: 155,
  splitThreshold_s: 7.5,
};
Intuition

Dialogue looks rushed fast, so keep utilization lower.

Narration can be denser because visuals can do more lifting.

2/6 — Compute “speech seconds” (prefer real TTS, fallback estimate)

// speechBudget.speechSeconds.ts
export function estimateSpeechSeconds(text: string, wpm = 155): number {
  const words = (text.trim().match(/\S+/g) ?? []).length;
  const base = (words / wpm) * 60;

  // tiny pause proxy
  const punct = (text.match(/[,\.\!\?\:\;—-]/g) ?? []).length;
  const pause = punct * 0.015;

  return round2(base + pause);
}

function round2(n: number) {
  return Math.round(n * 100) / 100;
}

// Prefer real TTS durations if you have them
export function getSpeechSeconds(opts: {
  beatId: string;
  narrationText: string;
  ttsDurationsByBeatId?: Record<string, number>; // seconds
  wpmFallback?: number;
}): number {
  const exact = opts.ttsDurationsByBeatId?.[opts.beatId];
  if (typeof exact === "number" && exact > 0) return exact;
  return estimateSpeechSeconds(opts.narrationText, opts.wpmFallback ?? 155);
}
3/6 — Speech budget planner (turn speechSeconds → required clip seconds)

This is the core: requiredClipSeconds is computed from utilization + buffers.

// speechBudget.planner.ts
import { SpeechBudgetPolicy, DEFAULT_SPEECH_BUDGET, VoiceMode } from "./speechBudget.policy";
import { getSpeechSeconds } from "./speechBudget.speechSeconds";

export type BeatSpeechBudget = {
  beatId: string;
  voiceMode: VoiceMode;
  speechSeconds: number;
  currentClipSeconds: number;
  requiredClipSeconds: number;
  utilizationNow: number;
  utilizationIfAdjusted: number;
  needsChange: boolean;
  suggestion?: "EXTEND_BEAT" | "SPLIT_BEAT" | "SHORTEN_TEXT" | "SWITCH_TO_NARRATOR" | "REGEN_SORA_LONGER";
};

export function planSpeechBudget(opts: {
  beats: Array<{ id: string; type: string; duration_s: number; narration?: string }>;
  voiceModeForBeat: (beatType: string) => VoiceMode; // derived from format pack + hybrid rules
  ttsDurationsByBeatId?: Record<string, number>;
  policy?: Partial<SpeechBudgetPolicy>;
}): BeatSpeechBudget[] {
  const p: SpeechBudgetPolicy = { ...DEFAULT_SPEECH_BUDGET, ...(opts.policy ?? {}) };

  return opts.beats.map((b) => {
    const mode = opts.voiceModeForBeat(b.type);
    const speechSeconds = getSpeechSeconds({
      beatId: b.id,
      narrationText: b.narration ?? "",
      ttsDurationsByBeatId: opts.ttsDurationsByBeatId,
      wpmFallback: p.estimateWpm,
    });

    const maxUtil = mode === "SORA_DIALOGUE" ? p.maxUtilization.SORA_DIALOGUE : p.maxUtilization.EXTERNAL_NARRATOR;

    // required = buffers + (speech / maxUtil)
    const requiredClipSeconds = round2(p.buffers.leadIn_s + p.buffers.tail_s + speechSeconds / maxUtil);

    const utilizationNow = b.duration_s > 0 ? round3(speechSeconds / b.duration_s) : 1;
    const utilizationIfAdjusted = round3(speechSeconds / requiredClipSeconds);

    const needsChange = requiredClipSeconds > b.duration_s + 0.05;

    let suggestion: BeatSpeechBudget["suggestion"] | undefined;
    if (needsChange) {
      if (mode === "EXTERNAL_NARRATOR") {
        suggestion = requiredClipSeconds > p.splitThreshold_s ? "SPLIT_BEAT" : "EXTEND_BEAT";
      } else {
        // dialogue is expensive to extend (Sora regen) + can look rushed
        if (requiredClipSeconds > p.clipLimits.maxDialogueClipSeconds) suggestion = "SWITCH_TO_NARRATOR";
        else suggestion = "REGEN_SORA_LONGER";
      }
    } else if (utilizationNow > maxUtil + 0.03) {
      suggestion = mode === "SORA_DIALOGUE" ? "SHORTEN_TEXT" : "EXTEND_BEAT";
    }

    return {
      beatId: b.id,
      voiceMode: mode,
      speechSeconds,
      currentClipSeconds: b.duration_s,
      requiredClipSeconds,
      utilizationNow,
      utilizationIfAdjusted,
      needsChange,
      suggestion,
    };
  });
}

function round2(n: number) { return Math.round(n * 100) / 100; }
function round3(n: number) { return Math.round(n * 1000) / 1000; }
4/6 — Apply speech budget: update IR durations + decide Sora seconds (cost-aware)

This is where “speech budget per generated clip” becomes real: we only ask Sora for longer clips when we must.

// speechBudget.apply.ts
import { SpeechBudgetPolicy, DEFAULT_SPEECH_BUDGET } from "./speechBudget.policy";
import { BeatSpeechBudget } from "./speechBudget.planner";

export type SpeechBudgetApplyResult = {
  updatedBeats: Array<{ id: string; duration_s: number; narration?: string; type: string }>;
  soraSecondsOverrides: Record<string, number>; // beatId -> seconds to request for Sora clip (if applicable)
  actions: Array<{ beatId: string; action: string; detail: string }>;
};

export function applySpeechBudget(opts: {
  beats: Array<{ id: string; type: string; duration_s: number; narration?: string }>;
  budgets: BeatSpeechBudget[];
  policy?: Partial<SpeechBudgetPolicy>;
  // whether this beat uses a reusable plate (cheap to extend)
  isReusablePlateBeat: (beatType: string) => boolean;
}): SpeechBudgetApplyResult {
  const p: SpeechBudgetPolicy = { ...DEFAULT_SPEECH_BUDGET, ...(opts.policy ?? {}) };

  const soraSecondsOverrides: Record<string, number> = {};
  const actions: SpeechBudgetApplyResult["actions"] = [];

  const updatedBeats = opts.beats.map((b) => {
    const budget = opts.budgets.find((x) => x.beatId === b.id);
    if (!budget || !budget.needsChange) return b;

    // 1) External narrator: extend beat duration (often free if using plates)
    if (budget.voiceMode === "EXTERNAL_NARRATOR") {
      const capped = Math.min(budget.requiredClipSeconds, p.clipLimits.maxNarratedBeatSeconds);
      actions.push({
        beatId: b.id,
        action: "EXTEND_BEAT",
        detail: Extend beat from ${b.duration_s}s → ${capped}s to fit narration.,
      });
      return { ...b, duration_s: capped };
    }

    // 2) Sora dialogue: either regen longer or switch mode (caller can handle switch)
    if (budget.voiceMode === "SORA_DIALOGUE") {
      const desired = budget.requiredClipSeconds;

      if (desired <= p.clipLimits.maxDialogueClipSeconds) {
        actions.push({
          beatId: b.id,
          action: "REGEN_SORA_LONGER",
          detail: Dialogue needs ${desired}s; request longer Sora clip.,
        });
        soraSecondsOverrides[b.id] = Math.ceil(desired); // round up for Sora request
        return { ...b, duration_s: desired };
      }

      actions.push({
        beatId: b.id,
        action: "SWITCH_TO_NARRATOR",
        detail: Dialogue would need ${desired}s (> ${p.clipLimits.maxDialogueClipSeconds}s). Switch this beat to narrated visuals or split.,
      });

      // leave duration as-is here; upstream can switch mode/split
      return b;
    }

    return b;
  });

  return { updatedBeats, soraSecondsOverrides, actions };
}
5/6 — Processes that go along with speech budgets (split / shorten / switch)

This is the operational playbook (what your system does when budgets don’t fit).

// speechBudget.processes.ts
export type BudgetProcess = "SPLIT_BEAT" | "SHORTEN_TEXT" | "SWITCH_TO_NARRATOR";

export function decideProcessForOverBudget(opts: {
  voiceMode: "EXTERNAL_NARRATOR" | "SORA_DIALOGUE";
  requiredClipSeconds: number;
  maxDialogueClipSeconds: number;
  splitThreshold_s: number;
}): BudgetProcess | null {
  if (opts.voiceMode === "EXTERNAL_NARRATOR") {
    if (opts.requiredClipSeconds >= opts.splitThreshold_s) return "SPLIT_BEAT";
    return null; // just extend
  }

  // Dialogue:
  if (opts.requiredClipSeconds > opts.maxDialogueClipSeconds) return "SWITCH_TO_NARRATOR";
  return null; // regen longer
}

// super simple split suggestion (you can do it smarter later)
export function splitTextInHalf(text: string): [string, string] {
  const parts = text.split(/(?<=[\.\!\?])\s+/);
  if (parts.length <= 1) {
    const mid = Math.floor(text.length / 2);
    return [text.slice(0, mid).trim(), text.slice(mid).trim()];
  }
  const mid = Math.ceil(parts.length / 2);
  return [parts.slice(0, mid).join(" ").trim(), parts.slice(mid).join(" ").trim()];
}
How this plays out

If a narrated beat is too long → split the narration into two beats, reuse the same plate.

If a dialogue beat is too long → switch to narrator (or split into multiple dialogue beats, but that costs more Sora time).

6/6 — Wire into orchestrator: “speech budget pass” happens BEFORE Sora generation

// orchestrate.withSpeechBudget.ts (high-level)
import { planSpeechBudget } from "./speechBudget.planner";
import { applySpeechBudget } from "./speechBudget.apply";
import { DEFAULT_SPEECH_BUDGET } from "./speechBudget.policy";

export async function orchestrate(opts: any) {
  // A) Build IR (beats have narration text)
  let ir = opts.makeIR(opts.trend, opts.brief);

  // B) If using external narration: generate per-beat TTS first so we have REAL durations
  // (Optional but recommended; otherwise we use WPM estimate.)
  const ttsDurationsByBeatId = opts.ttsDurationsByBeatId ?? undefined;

  // C) Compute speech budgets per beat
  const budgets = planSpeechBudget({
    beats: ir.beats,
    voiceModeForBeat: (beatType: string) => opts.voiceModeForBeat(beatType), // from format+hybrid rules
    ttsDurationsByBeatId,
    policy: DEFAULT_SPEECH_BUDGET,
  });

  // D) Apply budgets: extend beats and generate Sora seconds overrides (cost-aware)
  const applied = applySpeechBudget({
    beats: ir.beats,
    budgets,
    policy: DEFAULT_SPEECH_BUDGET,
    isReusablePlateBeat: (beatType) => beatType === "STEP",
  });

  ir = { ...ir, beats: applied.updatedBeats };

  // E) Build shot plan using updated beat durations + overrides
  let shotPlan = opts.makeShotPlan({ ir });

  // apply Sora seconds overrides per beatId
  for (const s of shotPlan.shots) {
    const override = applied.soraSecondsOverrides[s.fromBeatId];
    if (override) s.seconds = override;
  }

  // F) Run Sora, then Remotion render plan as usual
  const assets = await opts.runSora({ shotPlan });
  const renderPlan = opts.makeRenderPlan({ ir, assets });

  return {
    ir,
    budgets,
    speechBudgetActions: applied.actions,
    shotPlan,
    assets,
    renderPlan,
  };
}
What “speech budget per generated clip” gives you (in practice)

For each beat/clip you now have:

speechSeconds (real TTS or estimated)

maxUtilization (how rushed you allow it to feel)

requiredClipSeconds (how long the clip must be)

suggested process if it doesn’t fit:

extend beat (cheap if plates)

regen Sora longer (costly)

split beat (usually best)

switch to narrator (best for long “dialogue”)

Recommended defaults for your factory

Explainers: EXTERNAL_NARRATOR, maxUtil ~ 0.78

Dialogue formats: SORA_DIALOGUE, maxUtil ~ 0.60–0.62, cap clip ≤ 6s
